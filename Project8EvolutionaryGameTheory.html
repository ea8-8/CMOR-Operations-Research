import numpy as np
import matplotlib.pyplot as plt

# Defining all necessary functions
def evo(M, N, b, rnd):
    # inputs: M = rows, N = columns, b = temptation, rnd = num rounds
    A = np.ones((M, N)) # create the initial matrix for the game
    A[M//2, N//2] = 0 # the middle cell is the defector
    
    frac = [] # will eventuallt store the fraction of cooperators / round
    tenthrnd = 9
    hundrnd = 99
    
    for i in range(rnd): # computing the fraction of cooperation then displaying it on a plot
        frac.append(np.sum(A)/(M * N)) # adding fraction of cooperators to the list
        Ae, S = score(A, b) # calculating the score 
        An = advance(S,A,Ae) # calculating the next generation
        
        if i == tenthrnd or i == hundrnd or i == rnd-1:
            print(f"Round {i}:")
            plt.title("M = " + str(M) + ", N = " + str(N) + ", rounds = " + str(i+1) + ", Cooperators vs Time")
            evodisp(A,An)
        
        A = An # so we can make the current gen the next gen, and keep the iteration going
    
    plt.figure() # start plotting the fraction of cooperators as time goes on
    plt.plot(frac)
    plt.title("M = " + str(M) + ", N = " + str(N))
    plt.ylabel("Fraction number of Cooperators")
    plt.xlabel("Rounds")
    plt.show() # showing the copperations over time plot
    # outputs = nothing other than the plot 

def score(A,b):
    # inputs = A =current board, b = temptation of the defectors
    M, N = A.shape
    Ae = np.zeros((M+2,N+2))
    Ae[1:M+1,1:N+1] = A

    # wrap edges
    Ae[0, 1:N+1] = A[M-1, :]
    Ae[M+1, 1:N+1] = A[0, :]
    Ae[1:M+1, 0] = A[:, N-1]
    Ae[1:M+1, N+1] = A[:, 0]
    
    # wrap corners
    Ae[0, 0] = A[M-1, N-1]
    Ae[0, N+1] = A[M-1, 0]
    Ae[M+1, 0] = A[0, N-1]
    Ae[M+1, N+1] = A[0, 0]

    S = np.zeros((M, N)) # initialize the score matrix

    for i in range(1, M+1):
        for j in range(1, N+1):
            se = Ae[i, j]    
            s = 0
            # iterate over 3x3 neighborhood around (i, j)
            for di in range(-1, 2):
                for dj in range(-1, 2):
                    val = Ae[i + di, j + dj]
                    if se == 1 and val == 1: # points from the prisoner's dilemma
                        s = s + 1
                    elif se == 0 and val == 1:
                        s = s + b
            S[i-1, j-1] = s
    return Ae, S # outputs = Ae = wrapped board, S = matrix that holds scores

def advance(S, A, Ae):
    # inputs = S = matrix that holds scores, A = current board for the game, Ae = wrapped board
    M, N = A.shape
    Sx = np.zeros((M+2, N+2))
    
    Sx[1:M+1, 1:N+1] = S
    # wrap edges
    Sx[0, 1:N+1] = S[M-1, :]
    Sx[M+1, 1:N+1] = S[0, :]
    Sx[1:M+1, 0] = S[:, N-1]
    Sx[1:M+1, N+1] = S[:, 0]
    # wrap corners
    Sx[0, 0] = S[M-1, N-1]
    Sx[0, N+1] = S[M-1, 0]
    Sx[M+1, 0] = S[0, N-1]
    Sx[M+1, N+1] = S[0, 0]
    An = np.zeros(np.shape(A))

    for i in range(1, M+1):
        for j in range(1, N+1):
            adj_scores = Sx[i-1:i+2, j-1:j+2]
            max = np.max(adj_scores)
            [r, c] = np.where(adj_scores == max)
            A2 = Ae[i-1:i+2, j-1:j+2]
            An[i-1, j-1] = A2[r[0], c[0]]
    return An # outputs = An = new board for the next round/gen

def evodisp(A, An): # essentially visualizes the change in strategies from A to An by assigning colors (b, r, y, g)
    # inputs = A = old board, An = new board
    M, N = A.shape
    D = np.zeros((M, N, 3)) # initialize D which is the 3x3 display matrix
    for i in range(M): # assigning colors
        for j in range(N):
            if A[i, j] == 1 and An[i, j] == 1: # cooperator stays cooperator
                D[i, j] = [0, 0, 1] # cooperator stays cooperator: blue
            elif A[i, j] == 0 and An[i, j] == 0: # defector stays defector
                D[i, j] = [1, 0, 0] # defector stays defector: red
            elif A[i, j] == 1 and An[i, j] == 0: # cooperator becomes defector
                D[i, j] = [1, 1, 0] # cooperator becomes defector: yellow
            elif A[i, j] == 0 and An[i, j] == 1: # defector becomes cooperator
                D[i, j] = [0, 1, 0] # defector becomes cooperator: green
    plt.imshow(D) # shows the differences in stategies on a plot
    plt.axis("equal")
    plt.show
    # outputs = none
